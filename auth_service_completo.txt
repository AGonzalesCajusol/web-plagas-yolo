 142:   String _generateUuidV4() {
 143:     final random = Random.secure();
 144:     final bytes = List<int>.generate(16, (_) => random.nextInt(256));
 145: 
 146:     bytes[6] = (bytes[6] & 0x0f) | 0x40;
 147:     bytes[8] = (bytes[8] & 0x3f) | 0x80;
 148: 
 149:     String hex(int value) => value.toRadixString(16).padLeft(2, '0');
 150:     final chars = bytes.map(hex).join();
 151: 
 152:     return '${chars.substring(0, 8)}-'
 153:         '${chars.substring(8, 12)}-'
 154:         '${chars.substring(12, 16)}-'
 155:         '${chars.substring(16, 20)}-'
 156:         '${chars.substring(20)}';
