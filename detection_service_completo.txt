  20:   Future<bool> saveDetection({
  21:     required String userId,
  22:     required String plagaNombre,
  23:     required double confianza,
  24:     required double? latitud,
  25:     required double? longitud,
  26:     required String ubicacionOrigen,
  27:     required Uint8List imageBytes,
  28:     Rect? boundingBox,
  29:     String? cultivoId,
  30:   }) async {
  31:     try {
  32:       final db = await LocalDB.instance.database;
  33:       final now = DateTime.now();
  34:       final fechaHora = now.toIso8601String();
  35:       final resolvedUserId = userId.trim();
  36:       if (resolvedUserId.isEmpty) return false;
  37: 
  38:       final resolvedCultivoId = await _resolveCultivoId(
  39:         db,
  40:         userId: resolvedUserId,
  41:         cultivoId: cultivoId,
  42:       );
  43:       if (resolvedCultivoId == null) return false;
  44: 
  45:       final rutaImagen = await _saveImageLocally(imageBytes);
  46: 
  47:       await db.transaction((txn) async {
  48:         final plagaId = await _getOrCreatePlagaId(txn, plagaNombre);
  49: 
  50:         await txn.insert('detecciones', {
  51:           'id': 'deteccion_${now.microsecondsSinceEpoch}',
  52:           'cultivo_id': resolvedCultivoId,
  53:           'plaga_id': plagaId,
  54:           'plaga_real_manual_id': null,
  55:           'confianza': confianza,
  56:           'fecha_hora': fechaHora,
  57:           'latitud': latitud,
  58:           'longitud': longitud,
  59:           'ubicacion_origen': ubicacionOrigen,
  60:           'box_left': boundingBox?.left,
  61:           'box_top': boundingBox?.top,
  62:           'box_right': boundingBox?.right,
  63:           'box_bottom': boundingBox?.bottom,
  64:           'ruta_imagen': rutaImagen,
  65:           'dispositivo_id': await _getDeviceName(),
  66:           'sincronizado': 0,
  67:           'cloud_id': null,
  68:           'sync_status': 'pendiente',
  69:           'last_sync_at': null,
  70:           'sync_error': null,
  71:           'retry_count': 0,
  72:         });
  73:       });
  74: 
  75:       return true;
  76:     } catch (error, stackTrace) {
  77:       // ignore: avoid_print
  78:       developer.log(
  79:         'Error guardando deteccion: $error',
  80:         name: 'DetectionService',
  81:         error: error,
  82:         stackTrace: stackTrace,
  83:       );
  84:       return false;
  85:     }
  86:   }

 117:   Future<bool> deleteDetection({
 118:     required String userId,
 119:     required String detectionId,
 120:   }) async {
 121:     final db = await LocalDB.instance.database;
 122:     final normalizedUserId = userId.trim();
 123:     final normalizedDetectionId = detectionId.trim();
 124: 
 125:     if (normalizedUserId.isEmpty || normalizedDetectionId.isEmpty) {
 126:       return false;
 127:     }
 128: 
 129:     final rows = await db.rawQuery('''
 130:     SELECT ruta_imagen
 131:     FROM detecciones
 132:     WHERE id = ?
 133:       AND cultivo_id IN (
 134:         SELECT id FROM cultivos WHERE usuario_id = ?
 135:       )
 136:     LIMIT 1
 137:   ''', [normalizedDetectionId, normalizedUserId]);
 138: 
 139:     if (rows.isEmpty) return false;
 140: 
 141:     final imagePath = rows.first['ruta_imagen']?.toString();
 142: 
 143:     final deletedRows = await db.delete(
 144:       'detecciones',
 145:       where: '''
 146:       id = ?
 147:       AND cultivo_id IN (
 148:         SELECT id FROM cultivos WHERE usuario_id = ?
 149:       )
 150:     ''',
 151:       whereArgs: [normalizedDetectionId, normalizedUserId],
 152:     );
 153: 
 154:     if (deletedRows > 0) {
 155:       await _deleteImageIfExists(imagePath);
 156:       return true;
 157:     }
 158: 
 159:     return false;
 160:   }

 162:   Future<int> clearUserDetections({
 163:     required String userId,
 164:   }) async {
 165:     final db = await LocalDB.instance.database;
 166:     final normalizedUserId = userId.trim();
 167: 
 168:     if (normalizedUserId.isEmpty) {
 169:       return 0;
 170:     }
 171: 
 172:     final rows = await db.rawQuery('''
 173:     SELECT ruta_imagen
 174:     FROM detecciones
 175:     WHERE cultivo_id IN (
 176:       SELECT id FROM cultivos WHERE usuario_id = ?
 177:     )
 178:   ''', [normalizedUserId]);
 179: 
 180:     final imagePaths = rows
 181:         .map((row) => row['ruta_imagen']?.toString())
 182:         .where((path) => path != null && path.trim().isNotEmpty)
 183:         .cast<String>()
 184:         .toList();
 185: 
 186:     final deletedCount = await db.delete(
 187:       'detecciones',
 188:       where: '''
 189:       cultivo_id IN (
 190:         SELECT id FROM cultivos WHERE usuario_id = ?
 191:       )
 192:     ''',
 193:       whereArgs: [normalizedUserId],
 194:     );
 195: 
 196:     if (deletedCount > 0) {
 197:       for (final imagePath in imagePaths) {
 198:         await _deleteImageIfExists(imagePath);
 199:       }
 200:     }
 201: 
 202:     return deletedCount;
 203:   }

 205:   Future<String> _saveImageLocally(Uint8List imageBytes) async {
 206:     final directory = await getApplicationDocumentsDirectory();
 207:     final detectionsDirectory = Directory('${directory.path}/detecciones');
 208: 
 209:     if (!await detectionsDirectory.exists()) {
 210:       await detectionsDirectory.create(recursive: true);
 211:     }
 212: 
 213:     final fileName = 'deteccion_${DateTime.now().microsecondsSinceEpoch}.jpg';
 214:     final imageFile = File('${detectionsDirectory.path}/$fileName');
 215:     await imageFile.writeAsBytes(imageBytes, flush: true);
 216: 
 217:     return imageFile.path;
 218:   }

 220:   Future<int> _getOrCreatePlagaId(Transaction txn, String plagaNombre) async {
 221:     final rows = await txn.query(
 222:       'plagas',
 223:       columns: ['id'],
 224:       where: 'nombre_comun = ? OR nombre_cientifico = ?',
 225:       whereArgs: [plagaNombre, plagaNombre],
 226:       limit: 1,
 227:     );
 228: 
 229:     if (rows.isNotEmpty) {
 230:       return rows.first['id'] as int;
 231:     }
 232: 
 233:     return txn.insert('plagas', {
 234:       'nombre_cientifico': plagaNombre,
 235:       'nombre_comun': plagaNombre,
 236:       'descripcion': 'Registro creado automaticamente desde una deteccion.',
 237:     });
 238:   }

 250:     final rows = await db.query(
 251:       'cultivos',

 253:       where: 'id = ? AND usuario_id = ?',

 255:       limit: 1,
 256:     );

 259:       return rows.first['id'] as String;

 265:   Future<void> _deleteImageIfExists(String? imagePath) async {
 266:     if (imagePath == null || imagePath.trim().isEmpty) return;
 267: 
 268:     try {
 269:       final file = File(imagePath);
 270:       if (await file.exists()) {
 271:         await file.delete();
 272:       }
 273:     } catch (error) {
 274:       // No detenemos la eliminaciÃ³n del registro si falla el borrado del archivo.
 275:       developer.log(
 276:         'No se pudo eliminar la imagen local.',
 277:         name: 'DetectionService',
 278:         error: error,
 279:       );
 280:     }
 281:   }
